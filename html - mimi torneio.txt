    <script>
      let btTimes = [];
      let btArtilheiros = [];

      // Ao carregar a página, verifica se há dados salvos
      window.onload = function() {
        btCarregarDados();
      };

      function btShuffle(arr) { return arr.sort(() => Math.random() - 0.5); }

      // Inicia um NOVO torneio (com sorteio)
      function btIniciarNovo() {
        const t1 = document.getElementById('bt-t1').value || "Time A";
        const t2 = document.getElementById('bt-t2').value || "Time B";
        const t3 = document.getElementById('bt-t3').value || "Time C";
        const t4 = document.getElementById('bt-t4').value || "Time D";
        
        btTimes = btShuffle([t1, t2, t3, t4]);
        btConstruirInterface();
        btSalvarDados(); // Salva estado inicial
      }

      // Constrói a tela de jogos baseada na ordem do array btTimes
      function btConstruirInterface() {
        // Preenche select
        const sel = document.getElementById('bt-select-time');
        sel.innerHTML = '';
        btTimes.forEach(t => { let o = document.createElement('option'); o.text = t; sel.add(o); });

        // Gera Rodadas
        const divContainer = document.getElementById('bt-lista-rodadas');
        divContainer.innerHTML = '';

        const criarHTML = (n, a, b, c, d) => {
          const html = `
          <div class="bt-rodada">
            <h4>Rodada ${n}</h4>
            <div class="bt-jogo">
              <span class="bt-nome-time">${a}</span>
              <div class="bt-placar-box">
                  <input type="number" class="bt-placar-input" data-t="${a}" onchange="btSalvarDados()"> x 
                  <input type="number" class="bt-placar-input" data-t="${b}" onchange="btSalvarDados()">
              </div>
              <span class="bt-nome-time">${b}</span>
            </div>
            <div class="bt-jogo">
              <span class="bt-nome-time">${c}</span>
              <div class="bt-placar-box">
                  <input type="number" class="bt-placar-input" data-t="${c}" onchange="btSalvarDados()"> x 
                  <input type="number" class="bt-placar-input" data-t="${d}" onchange="btSalvarDados()">
              </div>
              <span class="bt-nome-time">${d}</span>
            </div>
          </div>`;
          const el = document.createElement('div'); el.innerHTML = html; divContainer.appendChild(el);
        };
        
        criarHTML(1, btTimes[0], btTimes[3], btTimes[1], btTimes[2]);
        criarHTML(2, btTimes[0], btTimes[2], btTimes[3], btTimes[1]);
        criarHTML(3, btTimes[0], btTimes[1], btTimes[2], btTimes[3]);

        // Mostra telas
        document.getElementById('bt-setup').classList.add('bt-oculto');
        document.getElementById('bt-jogos').classList.remove('bt-oculto');
        document.getElementById('bt-classificacao').classList.remove('bt-oculto');
        document.getElementById('bt-botoes-finais').classList.remove('bt-oculto');
        
        btAtualizar(false); // Atualiza tabela sem salvar de novo para evitar loop
      }

      function btAtualizar(salvar = true) {
        let s = {}; btTimes.forEach(t => s[t] = {p:0,j:0,v:0,e:0,d:0,gp:0,gc:0,sg:0});
        
        document.querySelectorAll('.bt-jogo').forEach(j => {
          let inps = j.querySelectorAll('input');
          let v1 = inps[0].value, v2 = inps[1].value;
          if(v1!=='' && v2!==''){
             let g1=parseInt(v1), g2=parseInt(v2);
             let t1=inps[0].getAttribute('data-t');
             let t2=inps[1].getAttribute('data-t');
             s[t1].j++; s[t2].j++; s[t1].gp+=g1; s[t1].gc+=g2; s[t2].gp+=g2; s[t2].gc+=g1;
             s[t1].sg = s[t1].gp - s[t1].gc; s[t2].sg = s[t2].gp - s[t2].gc;
             if(g1>g2){ s[t1].p+=3; s[t1].v++; s[t2].d++; }
             else if(g2>g1){ s[t2].p+=3; s[t2].v++; s[t1].d++; }
             else { s[t1].p++; s[t2].p++; s[t1].e++; s[t2].e++; }
          }
        });
        
        let rank = Object.keys(s).map(k => ({n:k, ...s[k], perc: s[k].j>0 ? ((s[k].p/(s[k].j*3))*100).toFixed(0):0 }));
        rank.sort((a,b) => b.p-a.p || b.v-a.v || b.sg-a.sg || b.gp-a.gp);
        
        const tb = document.getElementById('bt-corpo-tabela'); tb.innerHTML = '';
        rank.forEach(t => {
          tb.innerHTML += `<tr><td>${t.n}</td><td style="background:#eee;font-weight:bold">${t.p}</td><td>${t.j}</td><td>${t.v}</td><td>${t.e}</td><td>${t.d}</td><td>${t.gp}</td><td>${t.gc}</td><td>${t.sg}</td><td>${t.perc}%</td></tr>`;
        });

        if(salvar) btSalvarDados();
      }

      // --- ARTILHARIA ---
      function btAddArtilheiro() {
         let n = document.getElementById('bt-nome-jogador').value;
         let t = document.getElementById('bt-select-time').value;
         let g = parseInt(document.getElementById('bt-gols-inicial').value);
         if(n){ 
           btArtilheiros.push({n,t,g}); 
           document.getElementById('bt-nome-jogador').value=''; 
           btRenderArt(); 
           btSalvarDados();
         }
      }
      function btMudaGol(i, d) { 
        btArtilheiros[i].g += d; 
        if(btArtilheiros[i].g<0) btArtilheiros[i].g=0; 
        btRenderArt(); 
        btSalvarDados();
      }
      function btRenderArt() {
         btArtilheiros.sort((a,b)=>b.g-a.g);
         const tb = document.getElementById('bt-lista-artilheiros'); tb.innerHTML='';
         btArtilheiros.forEach((a,i) => {
           tb.innerHTML += `<tr><td>${a.n}</td><td>${a.t}</td><td style="font-weight:bold">${a.g}</td><td class="bt-no-pdf"><button class="bt-btn-small bt-add" onclick="btMudaGol(${i},1)">+</button> <button class="bt-btn-small bt-rem" onclick="btMudaGol(${i},-1)">-</button></td></tr>`;
         });
      }

      // --- SISTEMA DE SALVAMENTO (LOCALSTORAGE) ---
      function btSalvarDados() {
        // Coletar placares na ordem
        let placares = [];
        document.querySelectorAll('.bt-placar-input').forEach(inp => {
          placares.push(inp.value);
        });

        const dados = {
          ativo: true,
          times: btTimes,
          artilheiros: btArtilheiros,
          placares: placares
        };
        
        localStorage.setItem('ipuTorneioDataV2', JSON.stringify(dados));
      }

      function btCarregarDados() {
        const salvos = localStorage.getItem('ipuTorneioDataV2');
        if (salvos) {
          const dados = JSON.parse(salvos);
          if (dados.ativo) {
            btTimes = dados.times;
            btArtilheiros = dados.artilheiros;
            
            // Reconstrói a tela sem sortear de novo
            btConstruirInterface();
            
            // Restaura os placares nos inputs
            const inputs = document.querySelectorAll('.bt-placar-input');
            if (dados.placares && dados.placares.length === inputs.length) {
              inputs.forEach((inp, index) => {
                inp.value = dados.placares[index];
              });
            }

            // Atualiza tudo
            btRenderArt();
            btAtualizar(false);
          }
        }
      }

      // --- GERAR PDF ---
      function btGerarPDF() {
         btSalvarDados(); // Garante que salvou antes de gerar
         const el = document.getElementById('blog-torneio-container');
         const noPdf = el.querySelectorAll('.bt-no-pdf, #bt-botoes-finais, #bt-setup');
         noPdf.forEach(x => x.style.display = 'none');

         const inputs = el.querySelectorAll('input.bt-placar-input');
         let replacements = [];

         inputs.forEach(input => {
            const span = document.createElement('span');
            span.innerText = input.value === '' ? '-' : input.value;
            span.className = 'bt-print-text';
            input.parentNode.insertBefore(span, input);
            input.style.display = 'none';
            replacements.push({input: input, span: span});
         });

         const opt = {
           margin: 5,
           filename: 'Torneio-Ipu-Resultados.pdf',
           image: { type: 'jpeg', quality: 0.98 },
           html2canvas: { scale: 2 },
           jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' }
         };

         html2pdf().set(opt).from(el).save().then(() => {
            noPdf.forEach(x => x.style.display = '');
            replacements.forEach(obj => {
               obj.span.remove();
               obj.input.style.display = '';
            });
         });
      }

      function btReiniciar() {
         if(confirm("Tem certeza? Isso apagará todos os dados atuais.")) {
           localStorage.removeItem('ipuTorneioDataV2');
           location.reload();
         }
      }
    </script>
